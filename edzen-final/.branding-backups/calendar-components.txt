function CalendarSidebar({ current, today, onDate, visible, setVisible, events, allEvents, onEvent, onCreate }: {
  current: Date; today: Date; onDate: (date: Date) => void;
  visible: Record<Category, boolean>; setVisible: React.Dispatch<React.SetStateAction<Record<Category, boolean>>>;
  events: EventItem[]; allEvents: EventItem[]; onEvent: (event: EventItem) => void; onCreate: () => void;
}) {
  const start = startOfWeek(startOfMonth(current), { weekStartsOn: 1 });
  const upcoming = events.filter((event) => event.date >= format(today, "yyyy-MM-dd")).slice(0, 3);
  return <aside className="space-y-6 p-4">
    <Button className="w-full" onClick={onCreate}><Icons.Plus />Create event</Button>
    <div>
      <div className="mb-3 flex items-center justify-between"><span className="text-sm font-semibold">{format(current, "MMMM yyyy")}</span><div className="flex"><Button variant="ghost" size="iconSm" onClick={() => onDate(addMonths(current, -1))} aria-label="Mini calendar previous month"><Icons.ChevronLeft /></Button><Button variant="ghost" size="iconSm" onClick={() => onDate(addMonths(current, 1))} aria-label="Mini calendar next month"><Icons.ChevronRight /></Button></div></div>
      <div className="grid grid-cols-7 gap-y-1 text-center">
        {["M", "T", "W", "T", "F", "S", "S"].map((day, index) => <span key={index} className="py-1 text-[10px] font-medium text-muted-foreground">{day}</span>)}
        {Array.from({ length: 42 }, (_, index) => addDays(start, index)).map((day) => <button key={format(day, "yyyy-MM-dd")} type="button" aria-label={format(day, "MMMM d, yyyy")} aria-pressed={isSameDay(day, current)} onClick={() => onDate(day)} className={cn("relative mx-auto grid size-7 place-items-center rounded-full text-[11px] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring", isSameDay(day, current) ? "bg-primary text-primary-foreground" : isSameDay(day, today) ? "bg-primary/10 font-semibold text-primary" : "hover:bg-muted", !isSameMonth(day, current) && "opacity-40")}>{format(day, "d")}{events.some((event) => event.date === format(day, "yyyy-MM-dd")) && !isSameDay(day, current) && <span className="absolute bottom-0.5 size-0.5 rounded-full bg-primary" />}</button>)}
      </div>
    </div>
    <div className="border-t pt-5"><div className="mb-3 flex items-center justify-between"><h2 className="text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">My calendars</h2><button type="button" onClick={() => setVisible(Object.fromEntries(categories.map((category) => [category, true])) as Record<Category, boolean>)} className="text-[11px] font-medium text-primary">Show all</button></div>
      <div className="space-y-0.5">{categories.map((category) => <label key={category} className="flex cursor-pointer items-center gap-2.5 rounded-md px-1 py-2 text-[12px]"><input type="checkbox" checked={visible[category]} onChange={() => setVisible((all) => ({ ...all, [category]: !all[category] }))} className="size-3.5 accent-primary" /><span className={cn("size-2 rounded-full", categoryDots[category])} /><span className="flex-1">{category}</span><span className="text-[10px] tabular-nums text-muted-foreground">{allEvents.filter((event) => event.category === category).length}</span></label>)}</div>
    </div>
    <div className="border-t pt-5"><h2 className="mb-3 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground">Coming up</h2>{upcoming.length ? <div className="space-y-3">{upcoming.map((event) => <button key={event.id} type="button" onClick={() => onEvent(event)} className="flex w-full items-start gap-2.5 text-start"><span className={cn("mt-1.5 size-2 shrink-0 rounded-full", categoryDots[event.category])} /><div className="min-w-0"><p className="truncate text-[12px] font-medium hover:text-primary">{event.title}</p><p className="mt-0.5 text-[10px] text-muted-foreground">{format(localDate(event.date), "MMM d")} · {event.allDay ? "All day" : event.start}</p></div></button>)}</div> : <p className="text-xs leading-5 text-muted-foreground">No upcoming events match your filters.</p>}</div>
  </aside>;
}

function CalendarToolbar({ current, view, setView, onNav, onToday, search, setSearch, onSidebar }: {
  current: Date; view: View; setView: (view: View) => void; onNav: (amount: number) => void; onToday: () => void;
  search: string; setSearch: (value: string) => void; onSidebar: () => void;
}) {
  const weekStart = startOfWeek(current, { weekStartsOn: 1 });
  const title = view === "Month" ? format(current, "MMMM yyyy") : view === "Day" ? format(current, "EEE, MMM d") : `${format(weekStart, "MMM d")} – ${format(addDays(weekStart, 6), "MMM d, yyyy")}`;
  return <div className="space-y-4 border-b px-4 py-4 sm:px-5">
    <div className="flex flex-wrap items-center justify-between gap-3"><div className="flex min-w-0 items-center gap-2"><Button variant="outline" size="iconSm" className="xl:hidden" onClick={onSidebar} aria-label="Open calendar filters"><Icons.PanelLeft /></Button><h2 className="text-lg font-semibold sm:text-xl">{title}</h2></div><div className="flex items-center gap-1.5"><Button variant="outline" size="sm" onClick={onToday}>Today</Button><Button variant="ghost" size="iconSm" onClick={() => onNav(-1)} aria-label="Previous calendar period"><Icons.ChevronLeft /></Button><Button variant="ghost" size="iconSm" onClick={() => onNav(1)} aria-label="Next calendar period"><Icons.ChevronRight /></Button></div></div>
    <div className="flex flex-wrap items-center justify-between gap-3"><div className="inline-flex rounded-lg border bg-muted/50 p-1" aria-label="Calendar view">{(["Month", "Week", "Day", "Agenda"] as View[]).map((item) => <button key={item} type="button" aria-pressed={view === item} onClick={() => setView(item)} className={cn("rounded-md px-3 py-1.5 text-xs font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring", view === item ? "bg-card text-primary shadow-sm" : "text-muted-foreground hover:text-foreground")}>{item}</button>)}</div>
      <label className="relative w-full sm:w-52"><span className="sr-only">Search calendar events</span><Icons.Search className="pointer-events-none absolute start-3 top-1/2 size-3.5 -translate-y-1/2 text-muted-foreground" /><input value={search} onChange={(event) => setSearch(event.target.value)} placeholder="Search events, people…" className="h-9 w-full rounded-lg border bg-card ps-9 pe-8 text-xs outline-none focus-visible:ring-2 focus-visible:ring-ring" />{search && <button type="button" aria-label="Clear calendar search" onClick={() => setSearch("")} className="absolute end-2 top-1/2 -translate-y-1/2 text-muted-foreground"><Icons.X className="size-3.5" /></button>}</label>
    </div>
  </div>;
}

function CalendarGrid({ view, current, today, events, onEvent, onEmpty, onShowDay }: {
  view: View; current: Date; today: Date; events: EventItem[]; onEvent: (event: EventItem) => void;
  onEmpty: (date: Date, start?: string) => void; onShowDay: (date: Date) => void;
}) {
  if (view === "Agenda") {
    const days = periodDays(current, view);
    return <div className="min-h-[540px] p-4 sm:p-5">{days.map((day) => {
      const items = events.filter((event) => event.date === format(day, "yyyy-MM-dd"));
      return <div key={format(day, "yyyy-MM-dd")} className="grid gap-3 border-b py-4 last:border-0 sm:grid-cols-[94px_1fr]"><div><p className={cn("text-sm font-semibold", isSameDay(day, today) && "text-primary")}>{format(day, "EEE, d")}</p><p className="mt-1 text-[11px] text-muted-foreground">{format(day, "MMMM")}</p></div><div className="space-y-2">{items.length ? items.map((event) => <button key={event.id} type="button" onClick={() => onEvent(event)} className="flex w-full flex-wrap items-center gap-3 rounded-lg border px-3 py-3 text-start transition-colors hover:bg-muted/30"><span className={cn("h-8 w-1 rounded-full", categoryDots[event.category])} /><div className="w-[100px] text-xs tabular-nums text-muted-foreground">{event.allDay ? "All day" : `${event.start} – ${event.end}`}</div><div className="min-w-0 flex-1"><p className="text-sm font-medium">{event.title}</p>{event.location && <p className="mt-1 flex items-center gap-1 text-[11px] text-muted-foreground"><Icons.MapPin className="size-3" />{event.location}</p>}</div><span className={cn("rounded-md px-2 py-1 text-[10px] font-medium", categoryStyles[event.category])}>{event.category}</span></button>) : <button type="button" onClick={() => onEmpty(day)} className="flex items-center gap-2 py-2 text-xs text-muted-foreground hover:text-primary"><Icons.Plus className="size-3.5" />No events · Add to your day</button>}</div></div>;
    })}</div>;
  }
  if (view !== "Month") return <CalendarTimeline current={current} today={today} view={view} events={events} onEvent={onEvent} onEmpty={onEmpty} />;
  const start = startOfWeek(startOfMonth(current), { weekStartsOn: 1 });
  return <div className="overflow-x-auto"><div className="grid min-w-[650px] grid-cols-7">
    {["Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"].map((day) => <div key={day} className="border-b bg-muted/20 px-3 py-3 text-[10px] font-semibold uppercase tracking-wide text-muted-foreground">{day.slice(0, 3)}</div>)}
    {Array.from({ length: 42 }, (_, index) => addDays(start, index)).map((day) => {
      const items = events.filter((event) => event.date === format(day, "yyyy-MM-dd"));
      return <section key={format(day, "yyyy-MM-dd")} className={cn("group min-h-[124px] border-b border-e p-2 last:border-e-0", !isSameMonth(day, current) ? "bg-muted/30" : day.getDay() === 0 || day.getDay() === 6 ? "bg-muted/10" : "bg-card")}>
        <div className="mb-2 flex items-center justify-between"><button type="button" onClick={() => onShowDay(day)} aria-label={`View ${format(day, "MMMM d")}`} className={cn("grid size-7 place-items-center rounded-full text-xs font-medium hover:ring-1 hover:ring-primary/30", isSameDay(day, today) ? "bg-primary text-primary-foreground" : !isSameMonth(day, current) ? "text-muted-foreground/70" : "text-foreground")}>{format(day, "d")}</button><button type="button" aria-label={`Create event on ${format(day, "MMMM d")}`} onClick={() => onEmpty(day)} className="grid size-6 place-items-center rounded-md text-muted-foreground opacity-0 hover:bg-muted focus:opacity-100 group-hover:opacity-100"><Icons.Plus className="size-3.5" /></button></div>
        <div className="space-y-1.5">{items.slice(0, 3).map((event) => <button key={event.id} type="button" onClick={() => onEvent(event)} title={`${event.title} · ${event.allDay ? "All day" : event.start}`} className={cn("flex w-full items-center gap-1.5 overflow-hidden rounded-md px-2 py-1.5 text-start text-[10px] font-medium transition-opacity hover:opacity-80", categoryStyles[event.category])}><span className={cn("size-1.5 shrink-0 rounded-full", categoryDots[event.category])} /><span className="truncate">{event.allDay ? "" : `${event.start} `}{event.title}</span></button>)}{items.length > 3 && <button type="button" onClick={() => onShowDay(day)} className="px-2 text-[10px] font-medium text-muted-foreground hover:text-primary">+{items.length - 3} more events</button>}</div>
      </section>;
    })}
  </div></div>;
}

function CalendarTimeline({ current, today, view, events, onEvent, onEmpty }: {
  current: Date; today: Date; view: "Week" | "Day"; events: EventItem[]; onEvent: (event: EventItem) => void; onEmpty: (date: Date, start?: string) => void;
}) {
  const viewport = useRef<HTMLDivElement>(null);
  useEffect(() => { if (viewport.current) viewport.current.scrollTop = 7 * 48; }, [view]);
  const days = periodDays(current, view);
  const grid = `64px repeat(${days.length}, minmax(0, 1fr))`;
  return <div ref={viewport} className="max-h-[660px] overflow-auto"><div className={view === "Week" ? "min-w-[820px]" : "min-w-[320px]"}>
    <div className="sticky top-0 z-20 grid border-b bg-card" style={{ gridTemplateColumns: grid }}><div className="border-e" />{days.map((day) => <div key={format(day, "yyyy-MM-dd")} className="border-e py-3 text-center"><p className="text-[10px] uppercase text-muted-foreground">{format(day, "EEE")}</p><span className={cn("mx-auto mt-1 grid size-8 place-items-center rounded-full text-sm font-semibold", isSameDay(day, today) && "bg-primary text-primary-foreground")}>{format(day, "d")}</span></div>)}</div>
    <div className="grid border-b" style={{ gridTemplateColumns: grid }}><div className="border-e px-2 py-3 text-[10px] text-muted-foreground">All day</div>{days.map((day) => <div key={format(day, "yyyy-MM-dd")} className="space-y-1 border-e p-2">{events.filter((event) => event.allDay && event.date === format(day, "yyyy-MM-dd")).map((event) => <button key={event.id} type="button" onClick={() => onEvent(event)} className={cn("w-full truncate rounded-md px-2 py-1 text-start text-[10px]", categoryStyles[event.category])}>{event.title}</button>)}</div>)}</div>
    <div className="grid" style={{ gridTemplateColumns: grid }}><div>{Array.from({ length: 24 }, (_, hour) => <div key={hour} className="h-12 border-b border-e px-2 pt-1 text-[10px] tabular-nums text-muted-foreground">{`${hour.toString().padStart(2, "0")}:00`}</div>)}</div>
      {days.map((day) => <div key={format(day, "yyyy-MM-dd")} className="relative h-[1152px] border-e">{Array.from({ length: 24 }, (_, hour) => <button key={hour} type="button" aria-label={`Create event ${format(day, "EEE, MMM d")} at ${hour}:00`} onClick={() => onEmpty(day, `${hour.toString().padStart(2, "0")}:00`)} className="block h-12 w-full border-b transition-colors hover:bg-primary/5 focus-visible:bg-primary/10 focus-visible:outline-none" />)}
        {layoutCalendarEvents(events.filter((event) => event.date === format(day, "yyyy-MM-dd"))).map(({ event, lane, lanes }) => <button key={event.id} type="button" onClick={() => onEvent(event)} title={`${event.title} · ${event.start} – ${event.end}`} style={{ top: `${minutes(event.start) * 0.8}px`, height: `${Math.max((minutes(event.end) - minutes(event.start)) * 0.8, 24)}px`, insetInlineStart: `calc(${(lane / lanes) * 100}% + 2px)`, width: `calc(${100 / lanes}% - 4px)` }} className={cn("absolute z-10 overflow-hidden rounded-md border border-current/15 px-2 py-1 text-start text-[10px] font-medium hover:brightness-95 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring", categoryStyles[event.category])}><p className="truncate font-semibold">{event.title}</p><p className="truncate opacity-80">{event.start} – {event.end}</p>{view === "Day" && event.location && <p className="mt-1 truncate opacity-80">{event.location}</p>}</button>)}
      </div>)}
    </div>
  </div></div>;
}
